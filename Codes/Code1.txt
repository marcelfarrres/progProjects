main {
    number firstNumber = 0;
    number secondNumber = 1;
    number fibNext = 0;
    number counter = 3;
    number maxLimit = 10;

    print("Fibonacci Series:");
    print(firstNumber);
    print(secondNumber);

    repeat_until (counter > maxLimit) {
        fibNext = firstNumber + secondNumber;
        print(fibNext);

        if (fibNext NOT= 0) {
            firstNumber = secondNumber;
            secondNumber = fibNext;
        } otherwise {
            print("Unexpected zero encountered");
        }

        counter = counter + 1;
    }
}
