<program> ::= <process-declarations> <main-declaration>

<process-declarations> ::= <process-declaration> <process-declarations> | <light-process-declaration> <process-declarations> | ε

<main-declaration> ::= main ocb <function-body> ccb

<process-declaration> ::= <data-type> <function-name> op <parameter-list> cp ocb <function-body> ccb

<light-process-declaration> ::= <light-function-name> op <parameter-list> cp ocb <function-body> ccb

<parameter-list> ::= <parameter> <parameter-tail> | ε

<parameter-tail> ::= , <parameter> <parameter-tail> | ε

<parameter> ::= <data-type> identifier

<function-body> ::= <statement> <function-body> | ε

<statement> ::= <variable-declaration> | identifier = <expression> ; | <light-function-call> | get_back <expression> ; | <iteration-statement> | <selection-statement> | <decision-statement> | <expression> ; | <special-function> ;

<variable-declaration> ::= <data-type> identifier = <literal> ;

<expression> ::= <logical-or-expression>

<logical-or-expression> ::= <logical-and-expression> <logical-or-prime>

<logical-or-prime> ::= || <logical-and-expression> <logical-or-prime> | ε

<logical-and-expression> ::= <equality-expression> <logical-and-prime>

<logical-and-prime> ::= && <equality-expression> <logical-and-prime> | ε

<equality-expression> ::= <relational-expression> <equality-prime>

<equality-prime> ::= <equality-operator> <relational-expression> <equality-prime> | ε

<equality-operator> ::= == | !=

<relational-expression> ::= <additive-expression> <relational-prime>

<relational-prime> ::= <relational-operator> <additive-expression> <relational-prime> | ε

<relational-operator> ::= > | < | >= | <=

<additive-expression> ::= <multiplicative-expression> <additive-prime>

<additive-prime> ::= <additive-operator> <multiplicative-expression> <additive-prime> | ε

<additive-operator> ::= + | -

<multiplicative-expression> ::= <unary-expression> <multiplicative-prime>

<multiplicative-prime> ::= <multiplicative-operator> <unary-expression> <multiplicative-prime> | ε

<multiplicative-operator> ::= * | / | %

<unary-expression> ::= <postfix-expression> | ++ <unary-expression> | -- <unary-expression> | - <unary-expression> | ! <unary-expression>

<postfix-expression> ::= <primary-expression> <postfix-prime>

<postfix-prime> ::= ++ <postfix-prime> | -- <postfix-prime> | ε

<primary-expression> ::= identifier | <literal> | op <expression> cp | <process-function-call>

<selection-statement> ::= if op <expression> cp ocb <statement> ccb | if op <expression> cp ocb <statement> ccb otherwise ocb <statement> ccb | decision_block op <expression> cp ocb <decision-statement> ccb

<iteration-statement> ::= repeat_until op <expression> cp ocb <statement> ccb

<decision-statement> ::= decision_block op <expression> cp ocb <case-list> <default-case> ccb

<case-list> ::= <case-statement> <case-list-tail> | ε

<case-list-tail> ::= <case-statement> <case-list-tail> | ε

<case-statement> ::= case <literal> : <case-body> break ;

<case-body> ::= <statement> <case-body> | ε

<default-case> ::= default : <case-body> break ;

<print-function> ::= print op <print-arguments> cp ;

<print-arguments> ::= <expression> <print-arguments-tail> | ε

<print-arguments-tail> ::= , <expression> <print-arguments-tail> | ε

<ascii-function> ::= ascii op <expression> cp

<scan-function> ::= scan op identifier cp

<special-function> ::= <ascii-function> | <print-function> | <scan-function>

<light-function-call> ::= <light-function-name> op <parameter-list> cp

<process-function-call> ::= <function-name> op <parameter-list> cp

<function-name> ::= process <literal>

<light-function-name> ::= light-process <literal>

<literal> ::= string | int | char | bool

<data-type> ::= number | boolean | character | word

<logical-operator> ::= && | || | !
